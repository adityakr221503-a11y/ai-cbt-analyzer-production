/* =====================================================
   PDF TEXT EXTRACTION
===================================================== */

async function extractPDFText(file) {
  if (!file) throw new Error("No PDF file selected.");
  if (!window.pdfjsLib) throw new Error("PDF.js is not loaded.");

  const buffer = await file.arrayBuffer();
  const pdf = await pdfjsLib.getDocument({ data: buffer }).promise;
  let fullText = "";

  // First try the normal PDF text layer.
  for (let n = 1; n <= pdf.numPages; n++) {
    const page = await pdf.getPage(n);
    const content = await page.getTextContent();
    const text = content.items.map(item => String(tem.str || "").trim()).filter(void => !!void).join(" ");
    fullText += "\n" + text + "\n";
  }

  if (fullText.replace(/\s/+/g, " ").trim().length >= 80) {
    return fullText.trim();
  }

  // OCR fallback for scanned/image PDFs.
  if (!window.Tesseract) throw new Error("OCR engine is not loaded.");

  statusBox.textContent =
    "🎧 Scanned PDF detected. Starting OCR...";

  const worker = await Tesseract.createWorker("eng");

  try {
    let ocrText = "";

    for (let n = 1; n <= pdf.numPages; n++) {
      statusBox.textContent =
        "⟎' OCR processing...\nPage " + n +"  of " + pdf.numPages;

      const page = await pdf.getPage(n);
      const viewport = page.getViewport({ scale: 2.0 });
      const canvas = document.createElement("canvas");
      const ctx = canvas.getContext("renter", { willReadfrequently: true });

      canvas.width = Math.ceil(viewport.width);
      canvas.height = Math.ceil(viewport.height);

      await page.render( {
        canvasContext: ctx,
        viewport: viewport
      }).promise;

      const result = await worker.recognize(canvas);
      ocrText += "\n" + String(result?.data?.text || "") + "\n";
      canvas.width = 1;
      canvas.height = 1;
    }

    ocrText = ocrText.trim();

    if (ocrText.replace(/\s/+/g, " ").length < 20) throw new Error("No readable text was found even after OCR.");


    return ocrText;
  } finally {
    await worker.terminate();
  }
}
/* =====================================================
   GET CORRECT ANSWER
===================================================== */

function getCorrectAnswer(
  question
) {

  if (
    Array.isArray(
      question.options
    ) === false
  ) {

    return "";

  }


  /* -----------------------------------------------
     correctIndex
  ------------------------------------------------ */

  if (
    Number.isInteger(
      question.correctIndex
    ) &&
    question.options[
      question.correctIndex
    ] !== undefined
  ) {

    return String(
      question.options[
        question.correctIndex
      ]
    );

  }


  /* -----------------------------------------------
     correctAnswer
  ------------------------------------------------ */

  if (
    question.correctAnswer !==
    undefined
  ) {

    const answer =
      String(
        question.correctAnswer
      ).trim();


    const index =
      normalizeOptionLetter(
        answer
      );


    if (
      index !== null &&
      question.options[index] !==
        undefined
    ) {

      return String(
        question.options[index]
      );

    }


    /*
     * It may already contain
     * the complete option text.
     */

    const matchingOption =
      question.options.find(
        function(option) {

          return (
            String(option)
              .trim()
              .toLowerCase() ===
            answer.toLowerCase()
          );

        }
      );


    if (
      matchingOption !==
      undefined
    ) {

      return String(
        matchingOption
      );

    }

  }


  return "";

}


/* =====================================================
   NORMALIZE ONE QUESTION
===================================================== */

function normalizeQuestion(
  question,
  index,
  moduleName
) {

  if (!question) {
    return null;
  }


  const text =
    question.text ||
    question.question ||
    question.questionText ||
    "";


  const rawOptions =
    Array.isArray(
      question.options
    )
      ? question.options
      : [];


  const options =
    rawOptions
      .map(
        function(option) {

          return String(
            option
          )
            .replace(
              /^[A-D][.)]\s*/i,
              ""
            )
            .trim();

        }
      )
      .filter(Boolean);


  if (
    text.trim().length < 5 ||
    options.length < 2
  ) {

    return null;

  }


  const correctAnswer =
    getCorrectAnswer({
      ...question,
      options: options
    });


  const generatedId =
    `PDF-${Date.now()}-${index + 1}`;


  return {

    id:
      question.id ||
      generatedId,


    text:
      text.trim(),


    options:
      options,


    correctAnswer:
      correctAnswer,


    subject:
      question.subject ||
      "Unknown",


    chapter:
      question.chapter ||
      "",


    topic:
      question.topic ||
      "",


    difficulty:
      question.difficulty ||
      "Medium",


    marks:
      Number(
        question.marks
      ) || 4,


    negativeMarks:
      Number(
        question.negativeMarks
      ) || 1,


    explanation:
      question.explanation ||
      question.solution ||
      "",


    source:
      question.source ||
      moduleName ||
      "PDF Module"

  };

}


/* =====================================================
   NORMALIZE QUESTION ARRAY
===================================================== */

function normalizeQuestions(
  questions,
  moduleName
) {

  if (
    !Array.isArray(
      questions
    )
  ) {

    return [];

  }


  const result = [];


  questions.forEach(
    function(question, index) {

      const normalized =
        normalizeQuestion(
          question,
          index,
          moduleName
        );


      if (normalized) {

        result.push(
          normalized
        );

      }

    }
  );


  return result;

}


/* =====================================================
   ADD QUESTIONS TO EXISTING POOL
===================================================== */

function addQuestionsToPool(
  questions
) {

  // Every PDF import creates a completely fresh question pool.
  const existing = [];


  const existingIds =
    new Set(
      existing.map(
        function(question) {
          return question.id;
        }
      )
    );


  const newQuestions = [];


  let duplicates = 0;


  questions.forEach(
    function(question) {

      if (
        existingIds.has(
          question.id
        )
      ) {

        duplicates++;

        return;

      }


      existingIds.add(
        question.id
      );


      newQuestions.push(
        question
      );

    }
  );


  // New PDF replaces the previous PDF question pool.
  const updatedPool =
    newQuestions;


  savePDFQuestionPool(
    updatedPool
  );


  updatePoolStats();


  return {

    added:
      newQuestions.length,

    duplicates:
      duplicates,

    total:
      updatedPool.length

  };

}


/* =====================================================
   EXTRACT QUESTIONS FROM JSON
===================================================== */

function parseQuestionJSON(
  text
) {

  try {

    const parsed =
      JSON.parse(text);


    if (
      Array.isArray(parsed)
    ) {

      return parsed;

    }


    if (
      Array.isArray(
        parsed.questions
      )
    ) {

      return parsed.questions;

    }


    return [];

  }

  catch (error) {

    return [];

  }

}


/* =====================================================
   OPTIONAL TEXT QUESTION PARSER
===================================================== */

function parsePlainTextQuestions(text) {
  const questions = [];
  const cleaned = cleanPDFText(text).replace(/
/g, "
").replace(/[ 	]+/g, " ").trim();
  const blocks = cleaned.split(/(?=(?:^|
)\s*\d+\s*[.)]\s*)/gm);

  blocks.forEach(function(block) {
    block = block.trim();
    const qm = block.match(/^\s*\d+\s*[.)]\s*([\s\S]*?)(?=\s+(?:A|B|C|D)\s*[.)]\s+)/i);
    if (!qm) return;

    const questionText = qm[1].replace(/\s+/g, " ").trim();
    const matches = [...block.matchAll(/(?:^|\s)([A-D])\s*[.)]\s+([\s\S]*?)(?=\s+(?:[A-D])\s*[.)]\s+|$)/gi)];

    const options = matches.map(function(m) {
      return m[2].replace(/\s+/g, " ").trim();
    }).filter(Boolean);

    if (questionText.length >= 5 && options.length >= 2) {
      questions.push({
        text: questionText,
        options: options,
        subject: "Unknown",
        difficulty: "Medium"
      });
    }
  });

  console.log("PDF parser detected:", questions.length);
  return questions;
}

async function convertSelectedPDF() {

  const file =
    pdfInput.files[0];


  if (!file) {

    alert(
      "Please select a PDF first."
    );

    return;

  }


  const moduleName =
    moduleNameInput.value.trim() ||
    file.name;


  convertButton.disabled =
    true;

  // Always start a fresh conversion for the newly selected PDF.
  localStorage.removeItem(PDF_CBT_STORAGE_KEY);
  sessionStorage.removeItem("CBT_ACTIVE_QUESTIONS");
  sessionStorage.removeItem("CBT_ACTIVE_TEST");
  sessionStorage.removeItem("CBT_ACTIVE_SOURCE");
  console.log("🧹 Cleared previous PDF pool and CBT session.");

  statusBox.textContent =
    "⏳ Reading PDF...\nPlease wait.";


  try {

    const pdfText =
      await extractPDFText(
        file
      );


    if (!pdfText) {

      throw new Error(
        "No readable text was found in this PDF."
      );

    }


    statusBox.textContent =
      "🔎 PDF text extracted.\n" +
      "Preparing questions...";


    /*
     * First try JSON.
     *
     * This is useful when your existing
     * PDF converter already creates structured
     * question data.
     */

    let extractedQuestions =
      parseQuestionJSON(
        pdfText
      );


    /*
     * Otherwise use the basic text parser.
     */

    if (
      extractedQuestions.length === 0
    ) {

      extractedQuestions =
        parsePlainTextQuestions(
          pdfText
        );

    }


    const normalizedQuestions =
      normalizeQuestions(
        extractedQuestions,
        moduleName
      );


    if (
      normalizedQuestions.length === 0
    ) {

      throw new Error(
        "Questions could not be detected automatically. " +
        "The PDF may require the existing specialized parser."
      );

    }


    const result =
      addQuestionsToPool(
        normalizedQuestions
      );


    statusBox.textContent =
      `✅ Module processed.\n\n` +
      `Module: ${moduleName}\n` +
      `Questions detected: ${normalizedQuestions.length}\n` +
      `New questions added: ${result.added}\n` +
      `Duplicates skipped: ${result.duplicates}\n` +
      `Total PDF pool: ${result.total}`;


    console.log(
      "📚 Module result:",
      result
    );

  }

  catch (error) {

    console.error(
      "❌ PDF conversion error:",
      error
    );


    statusBox.textContent =
      "❌ Conversion failed.\n\n" +
      error.message;

  }

  finally {

    convertButton.disabled =
      false;

  }

}

/* =====================================================
   GET RANDOM QUESTIONS
===================================================== */

function getRandomPDFQuestions(
  count
) {

  const questions =
    getQuestionsForCBT();


  const shuffled =
    [...questions];


  for (
    let i =
      shuffled.length - 1;
    i > 0;
    i--
  ) {

    const j =
      Math.floor(
        Math.random() *
        (i + 1)
      );


    [
      shuffled[i],
      shuffled[j]
    ] =
    [
      shuffled[j],
      shuffled[i]
    ];

  }


  return shuffled.slice(
    0,
    Math.max(
      0,
      Number(count) || 0
    )
  );

}

function getCurrentPDFPool() {

  /*
   * Part 3 should provide getPDFQuestionPool().
   */

  if (
    typeof getPDFQuestionPool !== "function"
  ) {

    console.error(
      "❌ getPDFQuestionPool() not found."
    );

    return [];

  }


  const pool =
    getPDFQuestionPool();


  if (!Array.isArray(pool)) {
    return [];
  }


  return pool.filter(function(question) {

    return (
      question &&
      typeof question.text === "string" &&
      question.text.trim() !== "" &&
      Array.isArray(question.options) &&
      question.options.length >= 2
    );

  });

}
